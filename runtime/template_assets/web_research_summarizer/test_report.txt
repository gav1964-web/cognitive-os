from web_research_summarizer.contracts import ResearchReport, SummaryItem
from web_research_summarizer.report import write_report


def test_markdown_report_contains_single_summary_and_links(tmp_path):
    output = tmp_path / 'report.md'
    write_report(ResearchReport('query', [SummaryItem('Title', 'https://e.test/a', 'Summary')], []), str(output))
    text = output.read_text(encoding='utf-8')
    assert '## Summary' in text
    assert '## 1. Title' not in text
    assert '1. Title - https://e.test/a' in text


def test_json_report_contains_items(tmp_path):
    output = tmp_path / 'report.json'
    write_report(ResearchReport('query', [SummaryItem('Title', 'https://e.test/a', 'Summary')], []), str(output))
    assert '"items"' in output.read_text(encoding='utf-8')
