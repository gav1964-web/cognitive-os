def test_cli_popular_sites_mode_discovers_sources_filters_topic_and_writes_single_md_summary(tmp_path, monkeypatch):
    import news_site_scraper.cli as cli
    pages = {
        'https://duckduckgo.com/html/?q=Trump+news': """
        <a href='/l/?uddg=https%3A%2F%2Fwww.reuters.com%2Fworld%2Fus%2Ftrump-a'>Reuters</a>

def test_popular_topic_mode_falls_back_to_search_article_rows_when_homepages_miss_topic(tmp_path, monkeypatch):
    import news_site_scraper.cli as cli
    search_html = """
    <a href='/l/?uddg=https%3A%2F%2Fwww.reuters.com%2Fworld%2Feurope%2Fputin-peace-talks-2026-07-30%2F'>Putin comments on peace talks</a>
    <a href='/l/?uddg=https%3A%2F%2Fwww.bbc.com%2Fnews%2Fworld-europe-putin-security'>Putin security meeting in Moscow</a>
    """
    def fake_fetch(url, **kwargs):
        if url.startswith('https://duckduckgo.com/html/'):
            return search_html
        if url in {'https://www.reuters.com/', 'https://www.bbc.com/'}:
            return "<a href='/news/world'>World news</a><a href='/politics'>Politics</a>"
        return "<html><head><meta name='description' content='Article evidence about Putin and diplomatic talks.'></head></html>"
    monkeypatch.setattr(cli, 'fetch_html', fake_fetch)
    monkeypatch.setattr(cli, 'discover_news_sources_with_llm', lambda topic, **kwargs: ['https://www.reuters.com/', 'https://www.bbc.com/'])
    monkeypatch.setattr(cli, 'summarize_rows_with_llm', lambda rows: 'Общее резюме: материалы описывают заявления Путина и переговоры по безопасности.')
    output = tmp_path / 'putin.md'
    assert main(['popular', str(output), '--use-popular-sites', '--topic', 'Путин', '--source-discovery', 'search', '--top-per-site', '2', '--total-top', '2', '--summary-mode', 'llm', '--browser', 'off']) == 0
    text = output.read_text(encoding='utf-8')
    assert 'Общее резюме: материалы описывают заявления Путина' in text
    assert 'Putin comments on peace talks' in text
    assert '## Ссылки для проверки' in text


def test_topic_article_discovery_uses_bing_rss_when_search_html_has_no_results():
    from news_site_scraper.source_discovery import discover_news_article_rows
    rss = '''<?xml version="1.0"?><rss><channel>
    <item><title>Putin discusses security talks</title><link>https://www.reuters.com/world/europe/putin-security-talks-2026-07-30/</link><description>Search result evidence about Putin talks.</description></item>
    </channel></rss>'''
    calls = []
    def fake_fetch(url, **kwargs):
        calls.append(url)
        return rss if 'format=rss' in url else '<html><body>search form only</body></html>'
    rows = discover_news_article_rows('Путин', max_items=5, fetcher=fake_fetch)
    assert rows[0]['title'] == 'Putin discusses security talks'
    assert rows[0]['source'] == 'reuters.com'
    assert any('Putin+news' in url for url in calls)

def test_llm_source_discovery_recovers_urls_from_non_json_model_text(monkeypatch):
    import news_site_scraper.source_discovery as discovery
    class FakeResponse:
        headers = {}
        def __enter__(self):
            return self
        def __exit__(self, *args):
            return False
        def read(self):
            return b'{"choices":[{"message":{"content":"sources: https://www.reuters.com/world/us/trump-story, http://www.w3.org/TR/html4/ and https://apnews.com/article/trump-story"}}]}'
    monkeypatch.setattr(discovery, 'urlopen', lambda request, timeout=30: FakeResponse())
    assert discovery.discover_news_sources_with_llm('Trump') == ['https://www.reuters.com/', 'https://apnews.com/']

def test_cli_collects_multiple_sites_then_dedupes_and_writes_one_summary(tmp_path, monkeypatch):
    import news_site_scraper.cli as cli
    ixbt = """
    <a href='/news/2026/07/28/a.html'>Первая новость ixbt про процессоры</a>
    <a href='/news/2026/07/28/b.html'>Вторая новость ixbt про видеокарты</a>
    """
    d3 = """
    <a href='/1145827'>Первая новость 3dnews про ИИ</a>
    <a href='/1145814'>Вторая новость 3dnews про спутники</a>
    <a href='https://ixbt.com/news/2026/07/28/a.html'>Первая новость ixbt про процессоры</a>
    """
    article = "<html><head><meta name='description' content='Подробное описание статьи для единого суммари по нескольким источникам.'></head></html>"
    def fake_fetch(url):
        if url == 'https://ixbt.com/':
            return ixbt
        if url == 'https://3dnews.ru/':
            return d3
        return article
    monkeypatch.setattr(cli, 'fetch_html', fake_fetch)
    output = tmp_path / 'combined.md'
    assert main(['https://ixbt.com/', str(output), '--also-input', 'https://3dnews.ru/', '--top', '2']) == 0
    text = output.read_text(encoding='utf-8')
    assert '## Общее резюме' in text
    assert '## Ссылки для проверки' in text
    assert 'ixbt.com' in text and '3dnews.ru' in text
    assert text.count('https://ixbt.com/news/2026/07/28/a.html') == 1
