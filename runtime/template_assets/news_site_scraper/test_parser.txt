def test_parser_extracts_article_fixture_news():
    html = Path('tests/fixtures/news_page.html').read_text(encoding='utf-8')
    rows = parse_news_items(html, profile_for('https://ixbt.com/'))
    assert rows[0]['title'] == 'Тестовая fixture-новость 01 для проверки парсера ixbt-профиля'
    assert len(rows) == 10
    assert rows[0]['date'] == '2000-01-01'
    assert rows[0]['url'] == 'https://ixbt.com/news/fixture/parser-smoke-01.html'
    assert len(rows[0]['summary']) > 120
    assert rows[0]['source'] == 'ixbt.com'


def test_parser_handles_empty_or_malformed_html():
    assert parse_news_items('', profile_for('https://3dnews.ru/')) == []
    assert parse_news_items('<html><body>missing article</body></html>', profile_for('https://3dnews.ru/')) == []


def test_parser_extracts_news_link_fallback_and_deduplicates():
    html = "<a href='/news/2026/07/27/example.html'>Очень важная новость про технологии</a>" * 2
    rows = parse_news_items(html, profile_for('https://3dnews.ru/'))
    assert len(rows) == 1
    assert rows[0]['url'] == 'https://3dnews.ru/news/2026/07/27/example.html'
    assert rows[0]['date'] == '2026-07-27'


def test_parser_combines_article_and_link_fallback_for_live_front_pages():
    html = """
    <article><a href='/news/2026/a.html'>Первая новость из article</a><p>Подробное описание первой новости, которого достаточно для краткого содержания карточки.</p></article>

def test_parser_discovers_news_from_next_data_json():
    html = '''<script id="__NEXT_DATA__" type="application/json">{"props":{"items":[{"title":"РБК сообщил важную новость рынка","anons":"Короткое проверяемое описание новости из встроенного JSON.","publishDate":"Mon, 27 Jul 2026 15:50:05 +0300","link":{"url":"/politics/27/07/2026/6a6749e59a794771c6aebe2f"}}]}}</script>'''
    rows = parse_news_items(html, profile_for('https://www.rbc.ru/'))
    assert rows[0]['title'] == 'РБК сообщил важную новость рынка'
    assert rows[0]['url'] == 'https://www.rbc.ru/politics/27/07/2026/6a6749e59a794771c6aebe2f'
    assert rows[0]['summary'].startswith('Короткое проверяемое описание')

def test_feed_discovery_finds_declared_and_common_feeds():
    html = '<link rel="alternate" type="application/rss+xml" href="/ru/rss/articles/">'
    urls = discover_feed_urls(html, profile_for('https://habr.com/'))
    assert urls[0] == 'https://habr.com/ru/rss/articles/'

    assert 'https://habr.com/RSS/news.xml' in urls

def test_sitemap_discovery_and_parser_keep_article_like_same_host_urls():
    html = '<a href="https://www.ixbt.com/sitemap-news.xml">sitemap</a>'
    urls = discover_sitemap_urls(html, profile_for('https://www.ixbt.com/'))
    assert 'https://www.ixbt.com/sitemap.xml' in urls
    assert 'https://www.ixbt.com/sitemap-news.xml' in urls
    xml = '''<?xml version="1.0"?><urlset>
    <url><loc>https://www.ixbt.com/news/2026/07/28/novyj-servis-zapuschen.html</loc></url>
    <url><loc>https://www.ixbt.com/about</loc></url>
    <url><loc>https://external.example/news/2026/07/28/bad.html</loc></url>
    </urlset>'''
    rows = rows_from_sitemap(xml, profile_for('https://www.ixbt.com/'))
    assert [row['url'] for row in rows] == ['https://www.ixbt.com/news/2026/07/28/novyj-servis-zapuschen.html']
    assert rows[0]['date'] == '2026-07-28'

def test_llm_extractor_normalizes_and_filters_model_rows():
    html = """
    <a href='/news/2026/good.html'>Компания представила новый сервис</a>
    <a href='https://external.example/news/bad.html'>Внешняя ссылка не подходит</a>
    """
    def fake_caller(messages):
        assert 'Компания представила новый сервис' in messages[1]['content']
        return {
            'diagnosis': 'HTML links exist but deterministic URL profile is too strict.',
            'strategy_candidate': {'strategy_type': 'direct_evidence_items', 'reason': 'candidate links are article-like in evidence', 'items': [
                {'title': 'Компания представила новый сервис', 'url': '/news/2026/good.html', 'date': '2026-07-28', 'summary': 'Краткое описание из evidence.'},
                {'title': 'Внешняя ссылка не подходит', 'url': 'https://external.example/news/bad.html', 'summary': 'bad'},
                {'title': 'Читать далее', 'url': '/news/2026/more.html', 'summary': 'bad'},
            ]},
            'evidence_refs': ['link:/news/2026/good.html'],
            'risks': ['model may misclassify navigation links'],
        }
    rows = extract_news_with_llm(html, profile_for('https://example.test/'), requested_top=10, caller=fake_caller)
    assert rows == [{
        'title': 'Компания представила новый сервис',
        'url': 'https://example.test/news/2026/good.html',
        'date': '2026-07-28',
        'summary': 'Краткое описание из evidence.',
        'source': 'example.test',
    }]
    diagnosis = diagnose_extraction_failure(html, profile_for('https://example.test/'), requested_top=10, caller=fake_caller)
    assert diagnosis['artifact_type'] == 'SiteDiagnosis'
    assert diagnosis['strategy_candidate']['artifact_type'] == 'StrategyCandidate'
    assert diagnosis['strategy_candidate']['strategy_type'] == 'direct_evidence_items'
    assert diagnosis['invariants']['model_output_executed'] is False


def test_llm_summary_rejects_non_russian_model_output():
    def fake_caller(messages):
        assert 'одно общее резюме на русском языке' in messages[0]['content']
        return {'summary': 'CNN and Reuters report about Trump and Iran across several collected items.'}
    rows = [{'title': 'Trump and Iran talks continue', 'url': 'https://cnn.com/a', 'summary': '', 'source': 'cnn.com'}]
    assert summarize_rows_with_llm(rows, caller=fake_caller) == ''


def test_llm_summary_retries_after_non_russian_model_output():
    calls = []
    def fake_caller(messages):
        calls.append(messages[0]['content'])
        if len(calls) == 1:
            return {'summary': 'CNN and Reuters report about Trump and Iran across several collected items.'}
        return {'summary': 'Общее резюме: материалы описывают политические решения Trump и развитие конфликта вокруг Iran без разбивки по сайтам.'}
    rows = [{'title': 'Trump and Iran talks continue', 'url': 'https://cnn.com/a', 'summary': '', 'source': 'cnn.com'}]
    assert 'Общее резюме: материалы описывают' in summarize_rows_with_llm(rows, caller=fake_caller)
    assert 'повторная попытка' in calls[1]


def test_strategy_executor_runs_feed_candidate_with_validated_urls():
    strategy = {
        'strategy_type': 'feed_candidate',
        'feed_urls': ['/rss.xml', 'https://external.example/rss.xml'],
    }
    xml = '''\ufeff<?xml version="1.0"?><rss><channel><item><title>RSS нашла первую новость</title><link>https://example.test/news/1</link><description><![CDATA[<p>Описание новости. Читать далее</p>]]></description></item></channel></rss>'''
    calls = []
    def fake_fetch(url):
        calls.append(url)
        return xml
    rows = execute_strategy_candidate(strategy, profile_for('https://example.test/'), requested_top=1, fetcher=fake_fetch)
    assert calls == ['https://example.test/rss.xml']
    assert rows[0]['title'] == 'RSS нашла первую новость'
    assert rows[0]['summary'] == 'Описание новости.'


def test_strategy_executor_runs_json_candidate_through_same_validator():
    strategy = {'strategy_type': 'json_candidate', 'items': [
        {'title': 'JSON содержит новость', 'url': '/news/json-1', 'summary': 'Описание.'},
        {'title': 'External bad', 'url': 'https://external.example/news/json-2', 'summary': 'bad'},
    ]}
    rows = execute_strategy_candidate(strategy, profile_for('https://example.test/'), requested_top=5)
    assert [row['url'] for row in rows] == ['https://example.test/news/json-1']
